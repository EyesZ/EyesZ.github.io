{{- $src := .Destination -}} {{- $alt := .PlainText -}} {{- $r := "" -}} {{- if not (urls.Parse $src).IsAbs -}} {{- $r = or (.Page.Resources.GetMatch $src) (resources.Get $src) -}} {{- end -}} {{- if and $r (in (slice "png" "jpg" "jpeg" "webp" "gif") (lower (path.Ext $src | strings.TrimPrefix "."))) -}} {{/* Explicit width/height keep cumulative layout shift at zero. */}} {{- $max := 960 -}} {{- $display := cond (gt $r.Width $max) ($r.Resize (printf "%dx webp" $max)) $r -}} {{/* Every candidate is capped at the source width: upscaling only ever adds blur and bytes, and these sources top out around 1500px. */}} {{- $set := slice -}} {{- range (slice 640 $max) -}} {{- if lt . $r.Width -}} {{- $c := $r.Resize (printf "%dx webp" .) -}} {{- $set = $set | append (printf "%s %dw" $c.RelPermalink $c.Width) -}} {{- end -}} {{- end -}} {{- $native := cond (gt $r.Width $max) ($r.Resize (printf "%dx webp" $r.Width)) $display -}} {{- $set = $set | append (printf "%s %dw" $native.RelPermalink $native.Width) -}} {{- else -}} {{- end -}}
